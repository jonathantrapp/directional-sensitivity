cmake_minimum_required(VERSION 3.20)
project(DirectionalSensitivity VERSION 0.1.0 LANGUAGES C CXX)
option(DIRECTIONAL_BUILD_DRIVER "Build the experimental Windows KMDF driver" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  # CMake adds /EHsc globally. Remove it so only the core can disable exception
  # handling without MSVC option-override warnings; CLI/tests retain normal STL.
  string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
endif()
add_library(directional STATIC core/transform.cpp core/config.cpp)
target_include_directories(directional PUBLIC include)
add_library(directional_utility STATIC utility/profile_io.cpp calibration/calibration.cpp calibration/capture_session.cpp calibration/movement_timing.cpp)
target_link_libraries(directional_utility PUBLIC directional)
if(MSVC)
  target_compile_options(directional PRIVATE /W4 /WX /GR- /EHs-c-)
else()
  target_compile_options(directional PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
endif()
add_executable(replay tools/replay.cpp)
target_link_libraries(replay PRIVATE directional_utility)
add_executable(calibrate tools/calibrate.cpp)
add_executable(review_capture tools/review_capture.cpp)
target_link_libraries(review_capture PRIVATE directional_utility)
add_executable(score_profile tools/score_profile.cpp)
target_link_libraries(score_profile PRIVATE directional_utility)
target_link_libraries(calibrate PRIVATE directional_utility)
add_executable(utility_tests tests/utility_tests.cpp)
target_link_libraries(utility_tests PRIVATE directional_utility)
add_executable(capture_session_tests tests/capture_session_tests.cpp)
target_link_libraries(capture_session_tests PRIVATE directional_utility)
if(WIN32)
  add_executable(recorder windows/recorder.cpp)
  target_compile_definitions(recorder PRIVATE UNICODE _UNICODE)
  target_link_libraries(recorder PRIVATE directional_utility user32 gdi32)
  add_executable(preview windows/preview.cpp)
  target_compile_definitions(preview PRIVATE UNICODE _UNICODE)
  target_link_libraries(preview PRIVATE directional_utility user32 gdi32)
  add_executable(preview_tests tests/preview_tests.cpp)
  target_link_libraries(preview_tests PRIVATE directional)
  add_executable(driver_ctl windows/driver_ctl.cpp)
  target_link_libraries(driver_ctl PRIVATE directional_utility setupapi cfgmgr32)
  add_executable(packet_tests tests/packet_tests.cpp)
  target_link_libraries(packet_tests PRIVATE directional)
  add_executable(raw_capture windows/raw_capture.cpp)
  target_link_libraries(raw_capture PRIVATE directional user32)
  if(MSVC)
    target_compile_options(recorder PRIVATE /W4 /WX /EHsc /utf-8)
    target_compile_options(preview PRIVATE /W4 /WX /EHsc)
    target_compile_options(preview_tests PRIVATE /W4 /WX /EHsc)
    target_compile_options(driver_ctl PRIVATE /W4 /WX /EHsc)
    target_compile_options(raw_capture PRIVATE /W4 /WX /EHsc)
    target_compile_options(packet_tests PRIVATE /W4 /WX /EHsc)
  endif()
endif()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE directional)
add_executable(config_tests tests/config_tests.cpp)
target_link_libraries(config_tests PRIVATE directional)
add_executable(c_link_test tests/c_link_test.c)
target_link_libraries(c_link_test PRIVATE directional)
add_test(NAME c_link_test COMMAND c_link_test)
if(MSVC)
  target_compile_options(replay PRIVATE /W4 /WX /EHsc)
  target_compile_options(core_tests PRIVATE /W4 /WX /EHsc)
  target_compile_options(directional_utility PRIVATE /W4 /WX /EHsc)
  target_compile_options(calibrate PRIVATE /W4 /WX /EHsc)
  target_compile_options(review_capture PRIVATE /W4 /WX /EHsc)
  target_compile_options(score_profile PRIVATE /W4 /WX /EHsc)
  target_compile_options(utility_tests PRIVATE /W4 /WX /EHsc)
  target_compile_options(capture_session_tests PRIVATE /W4 /WX /EHsc)
  target_compile_options(config_tests PRIVATE /W4 /WX /EHsc)
  target_compile_options(c_link_test PRIVATE /W4 /WX)
endif()
enable_testing()
add_test(NAME core_tests COMMAND core_tests)
add_test(NAME utility_tests COMMAND utility_tests)
add_test(NAME score_contract COMMAND ${CMAKE_COMMAND}
  -DSCORE=$<TARGET_FILE:score_profile> -DBINARY=${CMAKE_BINARY_DIR}
  -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/score_contract.cmake)
add_test(NAME capture_session_tests COMMAND ${CMAKE_COMMAND}
  -DTEST=$<TARGET_FILE:capture_session_tests> -DREVIEW=$<TARGET_FILE:review_capture> -DBINARY=${CMAKE_BINARY_DIR}
  -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/capture_contract.cmake)
add_test(NAME config_tests COMMAND config_tests)
if(WIN32)
  add_test(NAME preview_tests COMMAND preview_tests)
  add_test(NAME recorder_contract COMMAND ${CMAKE_COMMAND}
    -DRECORDER=$<TARGET_FILE:recorder> -DBINARY=${CMAKE_BINARY_DIR}
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/recorder_contract.cmake)
  add_test(NAME packet_tests COMMAND packet_tests)
endif()
add_test(NAME cli_contract COMMAND ${CMAKE_COMMAND}
  -DREPLAY=$<TARGET_FILE:replay> -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}
  -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_contract.cmake)
if(DIRECTIONAL_BUILD_DRIVER)
  add_subdirectory(driver)
endif()
